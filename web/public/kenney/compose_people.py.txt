"""Compose office people from Kenney's layered character sheet: body, legs, torso, hair."""
import json
import sys
from PIL import Image

src, out, preview = sys.argv[1], sys.argv[2], sys.argv[3]
sheet = Image.open(src).convert("RGBA")
tile = lambda c, r: sheet.crop((c * 17, r * 17, c * 17 + 16, r * 17 + 16))
PEOPLE = {   # (col, row) per layer, picked from the labelled grid
    "trader":  [(0, 0), (3, 0), (11, 0), (19, 0)],
    "trader2": [(0, 1), (3, 0), (7, 0), (21, 4)],
    "manager": [(0, 1), (3, 0), (15, 1), (21, 8)],
    "boss":    [(0, 2), (3, 0), (15, 5), (23, 4)],
}
strip = Image.new("RGBA", (16 * len(PEOPLE), 16))
for i, (name, layers) in enumerate(PEOPLE.items()):
    im = Image.new("RGBA", (16, 16))
    for c, r in layers:
        im.alpha_composite(tile(c, r))
    strip.alpha_composite(im, (16 * i, 0))
strip.save(out)
big = strip.resize((strip.width * 10, strip.height * 10), Image.NEAREST)
bg = Image.new("RGBA", big.size, (185, 138, 91, 255)); bg.alpha_composite(big); bg.save(preview)
print(json.dumps(list(PEOPLE)))
